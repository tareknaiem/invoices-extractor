# Invoices Extractor — runtime dependencies (Streamlit Cloud / Linux + local).

# --- Core application -------------------------------------------------------
streamlit==1.64.0
pdfplumber==0.11.10
pandas==3.0.6
openpyxl==3.1.5

# numpy and PyYAML are capped by paddlex (pulled in by paddleocr): paddlex 3.7.x
# requires "numpy<2.4,>=1.24" and exactly "PyYAML==6.0.2". Pins that disagree
# with those caps make pip abort the whole install with a ResolutionImpossible
# conflict — reported by Streamlit Cloud as "Error installing requirements".
numpy==2.3.5
PyYAML==6.0.2

# Fast fuzzy column matching for the dynamic (unknown-supplier) fallback; the
# extractor degrades gracefully to the stdlib difflib when it is absent.
rapidfuzz==3.14.6

# --- OCR for scanned / Arabic invoices (optional at runtime) ----------------
# paddleocr ships a universal wheel; paddlepaddle ships manylinux wheels for
# CPython 3.9-3.13 only (no cp314 wheels exist yet), so the interpreter that
# runs `pip install -r requirements.txt` must be <=3.13. On Streamlit Cloud pick
# Python 3.13 in "Advanced settings" when deploying (version files such as
# .python-version are not honored by the platform). Without paddlepaddle the
# app still runs and reports OCR as unavailable; PDF page
# rendering uses pdfplumber's built-in pypdfium2 backend (no poppler needed).
paddleocr==3.7.0
paddlepaddle==3.3.1
