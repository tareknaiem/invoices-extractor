# Invoices Extractor — runtime dependencies (Streamlit Cloud / Linux + local).
#
# Install policy — prebuilt CPU wheels, fast on Linux deploy boxes:
#   * the official PaddlePaddle CPU wheel index is added as an extra source
#     (PyPI stays active) so paddlepaddle downloads from Paddle's CDN;
#   * paddlepaddle is wheel-only — it ships ~193MB prebuilt CPU wheels and has
#     no source distribution; a source build or a stalled oversized download
#     times out the Streamlit Cloud build;
#   * prefer wheel-backed versions for every dependency.
--extra-index-url https://www.paddlepaddle.org.cn/packages/stable/cpu/
--only-binary paddlepaddle
--prefer-binary

# --- Core application -------------------------------------------------------
streamlit==1.64.0
pdfplumber==0.11.10
pandas==3.0.6
openpyxl==3.1.5

# numpy and PyYAML are capped by paddlex (pulled in by paddleocr): paddlex 3.7.x
# requires "numpy<2.4,>=1.24" and exactly "PyYAML==6.0.2". Pins that disagree
# with those caps make pip abort the whole install with a ResolutionImpossible
# conflict — reported by Streamlit Cloud as "Error installing requirements".
numpy==2.3.5
PyYAML==6.0.2

# Fast fuzzy column matching for the dynamic (unknown-supplier) fallback; the
# extractor degrades gracefully to the stdlib difflib when it is absent.
rapidfuzz==3.14.6

# --- OCR for scanned / Arabic invoices (optional at runtime) ----------------
# paddleocr ships a universal wheel. paddlepaddle ships ~193MB prebuilt CPU
# wheels only (no sdist) for CPython 3.9-3.13 — no cp314 wheels exist on any
# index yet, so the deploy interpreter must be <=3.13: on Streamlit Cloud pick
# Python 3.13 in "Advanced settings" (version files such as .python-version
# are not honored by the platform). The >=3.3.1,<4 range makes pip take the
# current CPU wheels from the official index above (3.4.x on Linux/Windows)
# while older platforms fall back to 3.3.1 on PyPI. Without paddlepaddle the
# app still runs and reports OCR as unavailable; PDF page rendering uses
# pdfplumber's built-in pypdfium2 backend (no poppler needed).
paddleocr==3.7.0
paddlepaddle>=3.3.1,<4
